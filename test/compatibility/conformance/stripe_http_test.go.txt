package corpus_test

import (
	"context"
	"io"
	"net/http"
	"net/http/httptest"
	"sync/atomic"
	"testing"

	sdk "example.test/corpus/api"
	"example.test/corpus/api/types"
)

func TestStripeHTTPContract(t *testing.T) {
	var calls atomic.Int32
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		calls.Add(1)
		if r.Method != "GET" || r.URL.Path != "/v1/balance" || r.URL.RawQuery != "" || r.Header.Get("Authorization") != "Bearer fixture-token" {
			t.Error("unexpected request", r.Method, r.URL, r.Header)
		}
		w.Header().Set("Content-Type", "application/json")
		switch r.Header.Get("X-Case") {
		case "denied":
			w.WriteHeader(403)
			io.WriteString(w, `{"error":{"type":"invalid_request_error","message":"blocked"}}`)
		case "invalid":
			io.WriteString(w, `{"object":"wrong","available":[],"pending":[],"livemode":false}`)
		default:
			io.WriteString(w, `{"object":"balance","available":[],"pending":[],"livemode":false}`)
		}
	}))
	defer server.Close()
	client, err := sdk.NewClient(sdk.ClientOptions{BaseURL: server.URL})
	if err != nil {
		t.Fatal(err)
	}
	for _, scenario := range []string{"success", "denied", "invalid"} {
		options := types.RequestOptions{SecurityRequirement: sdk.GetBalanceSecurityBearerAuth, Authorization: types.Some("Bearer fixture-token"), Headers: http.Header{"X-Case": {scenario}}}
		value, err := client.Operations.GetBalance().WithOptions(options).Do(context.Background())
		if scenario == "success" {
			if err != nil {
				t.Fatal(err)
			}
			balance, present := value.As200JSON()
			if !present || balance.Livemode || len(balance.Available) != 0 || len(balance.Pending) != 0 || string(balance.Object) != "balance" {
				t.Fatal("balance response changed", balance, present)
			}
		} else if err == nil {
			t.Fatal("HTTP or response validation failure was lost", scenario)
		}
	}
	if calls.Load() != 3 {
		t.Fatal("missing HTTP request", calls.Load())
	}
}
